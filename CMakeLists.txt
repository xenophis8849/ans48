cmake_minimum_required(VERSION 3.18)
project(ans48 C CXX)
set(CMAKE_CXX_STANDARD 17)
# 编译产物里不留本机绝对路径（assert / __FILE__ 会把源文件路径写进二进制）
add_compile_options(-ffile-prefix-map=${CMAKE_SOURCE_DIR}=.)
# pffft（BSD 类许可，见 third_party/pffft/LICENSE.txt）：外圈的实数 FFT
add_library(pffft STATIC third_party/pffft/src/pffft.c third_party/pffft/src/pffft_common.c)
target_include_directories(pffft PUBLIC third_party/pffft/include/pffft)
set_target_properties(pffft PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(ANDROID AND (ANDROID_ABI MATCHES "arm"))
  target_compile_definitions(pffft PRIVATE PFFFT_ENABLE_NEON)
endif()
target_compile_options(pffft PRIVATE -O3 -ffast-math)

add_library(ans48 SHARED src/ans48.cpp src/ulunas_kernel.cpp src/models.cpp)
target_include_directories(ans48 PUBLIC include PRIVATE src)
target_link_libraries(ans48 PRIVATE pffft)
target_compile_options(ans48 PRIVATE -O3 -ffast-math -fvisibility=hidden)
if(ANDROID)
  target_link_options(ans48 PRIVATE -Wl,--gc-sections -s)
endif()
# 自检程序（主机默认编；Android 加 -DANS48_BUILD_CHECK=ON 也编，用来在手机上跑同一份自检）
if(ANS48_BUILD_CHECK OR NOT ANDROID)
  add_executable(ans48_check test/ans48_check.cpp)
  target_link_libraries(ans48_check PRIVATE ans48)
  add_executable(ans48_bench test/ans48_bench.cpp)
  target_link_libraries(ans48_bench PRIVATE ans48 pffft)
  add_executable(ans48_wrap test/ans48_wrap.cpp src/ulunas_kernel.cpp src/models.cpp)   # 直接测库内部的流式处理（不经 C 接口）
  target_include_directories(ans48_wrap PRIVATE src)
  target_link_libraries(ans48_wrap PRIVATE pffft)
  target_compile_options(ans48_wrap PRIVATE -O3 -ffast-math)
  add_executable(ans48_speccheck test/ans48_speccheck.cpp)   # 频谱档：按 aec48 联动契约自做分析 / 合成，对 Python 参照
  target_link_libraries(ans48_speccheck PRIVATE ans48 pffft)
  # 与 aec48 串联核对：只在给了 aec48 目录时编（两个库本身互不链接）
  if(ANS48_AEC48_DIR)
    if(ANDROID)
      set(AEC48_LIB ${ANS48_AEC48_DIR}/prebuilt/${ANDROID_ABI}/libaec48.so)
    else()
      set(AEC48_LIB ${ANS48_AEC48_DIR}/build-host/libaec48.so)
    endif()
    add_executable(ans48_aecchain test/ans48_aecchain.cpp)
    target_include_directories(ans48_aecchain PRIVATE ${ANS48_AEC48_DIR}/include)
    target_link_libraries(ans48_aecchain PRIVATE ans48 ${AEC48_LIB})
    if(NOT ANDROID)
      set_target_properties(ans48_aecchain PROPERTIES BUILD_RPATH "${ANS48_AEC48_DIR}/build-host")
    endif()
  endif()
  add_executable(ans48_robust test/ans48_robust.cpp)
  target_link_libraries(ans48_robust PRIVATE ans48)
  if(NOT ANDROID)
    target_link_libraries(ans48_robust PRIVATE pthread)
    # 内存分配失败注入：借 glibc 的 __libc_malloc 转发，只在能链接到它的 Linux 主机上编（macOS、musl 跳过）
    if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
      include(CheckCSourceCompiles)
      check_c_source_compiles("#include <stddef.h>
extern void* __libc_malloc(size_t);
int main(void) { return __libc_malloc(1) == 0; }" ANS48_HAVE_LIBC_MALLOC)
      if(ANS48_HAVE_LIBC_MALLOC)
        add_executable(ans48_oom test/ans48_oom.cpp)
        target_link_libraries(ans48_oom PRIVATE ans48)
      endif()
    endif()
  endif()
endif()

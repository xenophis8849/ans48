cmake_minimum_required(VERSION 3.18)
project(ans48 CXX)
set(CMAKE_CXX_STANDARD 17)
# 编译产物里不留本机绝对路径（assert / __FILE__ 会把源文件路径写进二进制）
add_compile_options(-ffile-prefix-map=${CMAKE_SOURCE_DIR}=.)
add_library(ans48 SHARED src/ans48.cpp src/ulunas_kernel.cpp src/models.cpp)
target_include_directories(ans48 PUBLIC include PRIVATE src)
target_compile_options(ans48 PRIVATE -O3 -ffast-math -fvisibility=hidden)
if(ANDROID)
  target_link_options(ans48 PRIVATE -Wl,--gc-sections -s)
endif()
# 自检程序（主机默认编；Android 加 -DANS48_BUILD_CHECK=ON 也编，用来在手机上跑同一份自检）
if(ANS48_BUILD_CHECK OR NOT ANDROID)
  add_executable(ans48_check test/ans48_check.cpp)
  target_link_libraries(ans48_check PRIVATE ans48)
  add_executable(ans48_bench test/ans48_bench.cpp)
  target_link_libraries(ans48_bench PRIVATE ans48)
  add_executable(ans48_wrap test/ans48_wrap.cpp src/ulunas_kernel.cpp src/models.cpp)   # 直接测库内部的流式处理（不经 C 接口）
  target_include_directories(ans48_wrap PRIVATE src)
  target_compile_options(ans48_wrap PRIVATE -O3 -ffast-math)
  add_executable(ans48_robust test/ans48_robust.cpp)
  target_link_libraries(ans48_robust PRIVATE ans48)
  if(NOT ANDROID)
    target_link_libraries(ans48_robust PRIVATE pthread)
  endif()
endif()
